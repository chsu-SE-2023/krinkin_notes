Наследование

```
class A:
	pass

class B(A): # Наследование
	def __init__(self):
		super().__init__() # Вызов инициализатора класса A
		super().func() # Вызов метода из A
	
	def __str__(self): # Перегрузка оператора приведения
		pass
```
