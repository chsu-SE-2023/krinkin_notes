; Начальные параметры
RD #50
WR 30 ; Адрес массива
IN
WR 31 ; Кол-во элементов
RD #0
WR 32 ; Итерация цикла


L:  RD @30

    ; Переход на следующий элемент
    RD 30
    ADD #1
    WR 30
    ; Увеличение итерации
    RD 32
    ADD #1
    WR 32

    ; Проверка итерации цикла
    RD 32
    SUB 31
    JS L

; Завершение работы
E: OUT
     HLT
