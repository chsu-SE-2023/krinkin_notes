set names cp866;

create database students character set cp866;

use students;

create table students (
    `Номер` integer auto_increment primary key, 
    `Фамилия` varchar(50), 
    `Год рождения` year, 
    `Математика` tinyint unsigned, 
    `Информатика` tinyint unsigned, 
    `Иностранный` tinyint unsigned
);

insert into students (`Фамилия`, `Год рождения`, `Математика`, `Информатика`, `Иностранный`)
values ('Кринкин', 2005, 3, 5, 5);

insert into students (`Фамилия`, `Год рождения`, `Математика`, `Информатика`, `Иностранный`)

create table students2 LIKE students;

select * from students2 order by `Фамилия`;

select `Фамилия`,`Год рождения` from students
    union
select `Фамилия`,`Год рождения` from students2;

select * from (
    select * from students
    union
    select * from students2
) as temp where `Математика` = 5;

select * from (
    select * from students
    union
    select * from students2
) as temp where `Математика` = 5 and `Информатика` = 5 and `Иностранный` = 5;

select * from (
    select * from students
    union
    select * from students2
) as temp where (year(curdate()) - `Год рождения`) >= 20;
